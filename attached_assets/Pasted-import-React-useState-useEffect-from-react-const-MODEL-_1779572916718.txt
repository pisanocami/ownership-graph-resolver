import React, { useState, useEffect } from 'react';

const MODEL = "claude-sonnet-4-20250514";

const SYSTEM_PROMPT = `You are a Revenue Inference Agent investigating private companies.

Your goal: estimate the annual revenue of a brand by gathering behavioral signals from the web. The user gives you a brand name; you investigate and produce a revenue range with confidence.

## Search strategy

1. First, disambiguate: confirm which entity you are investigating (find the official website, avoid name collisions with unrelated companies).
2. Then gather behavioral signals — proxies for revenue:
   - Web traffic estimates (SimilarWeb mentions in articles, third-party reports, Semrush quotes)
   - Hiring velocity (LinkedIn employee count, careers-page job postings, growth quotes)
   - Customer base size (review counts on G2, Trustpilot, Capterra, App Store, Amazon)
   - Pricing (public pricing pages — derive average deal size or ACV when possible)
   - Funding history (total raised, last round, valuation if mentioned)
   - Press coverage and growth indicators (year-over-year quotes from press)
   - Customer logos / case studies (enterprise customers signal higher ACVs)
   - Geographic footprint (countries served, fulfillment centers, regional offices)
   - Marketplace presence (Amazon Best Sellers Rank, app store ranking)
3. Cross-reference at least 2 sources per critical signal.
4. Be skeptical of any single revenue figure quoted online — they are often outdated, extrapolated, or marketing.

## Output

When you have gathered enough signals (aim for 5+ distinct signals, but stop earlier if more searches will not help), produce a final JSON block in this EXACT format:

\`\`\`json
{
  "revenue_estimate": {
    "low": <annual revenue lower bound in USD, integer>,
    "high": <annual revenue upper bound in USD, integer>,
    "central": <best point estimate in USD, integer>
  },
  "confidence": "low" | "medium" | "high",
  "signals_found": [
    {
      "type": "web_traffic" | "hiring" | "reviews" | "pricing" | "funding" | "customers" | "press" | "marketplace" | "other",
      "label": "<short signal name, max 5 words>",
      "value": "<the actual data point found>",
      "source": "<source name or short URL>",
      "weight": "low" | "medium" | "high"
    }
  ],
  "reasoning_summary": "<2-4 sentences explaining how the signals triangulate to the estimate>"
}
\`\`\`

Be honest: weak/contradictory signals → wider range, "low" confidence. Don't pretend precision you don't have. If you genuinely cannot find enough to estimate, set all numbers to 0 and confidence "low" with a reasoning_summary explaining why.`;

function App() {
  const [brand, setBrand] = useState("");
  const [hint, setHint] = useState("");
  const [loading, setLoading] = useState(false);
  const [trace, setTrace] = useState([]);
  const [result, setResult] = useState(null);
  const [error, setError] = useState(null);
  const [rawOutput, setRawOutput] = useState("");
  const [showRaw, setShowRaw] = useState(false);

  useEffect(() => {
    const link = document.createElement('link');
    link.href = 'https://fonts.googleapis.com/css2?family=Cormorant+Garamond:ital,wght@0,400;0,500;0,600;1,400;1,500&family=JetBrains+Mono:wght@400;500&family=Inter:wght@300;400;500;600&display=swap';
    link.rel = 'stylesheet';
    document.head.appendChild(link);
  }, []);

  async function investigate() {
    if (!brand.trim()) return;
    setLoading(true);
    setTrace([]);
    setResult(null);
    setError(null);
    setRawOutput("");

    try {
      const userMessage = `Investigate the revenue of: "${brand.trim()}"${hint.trim() ? `\n\nContext: ${hint.trim()}` : ""}`;

      const response = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: MODEL,
          max_tokens: 4096,
          system: SYSTEM_PROMPT,
          messages: [{ role: "user", content: userMessage }],
          tools: [
            { type: "web_search_20250305", name: "web_search", max_uses: 12 }
          ]
        })
      });

      if (!response.ok) {
        const errText = await response.text();
        throw new Error(`API ${response.status}: ${errText.slice(0, 240)}`);
      }

      const data = await response.json();

      const traceItems = [];
      let finalText = "";

      for (const block of (data.content || [])) {
        if (block.type === "text") {
          if (block.text && block.text.trim()) {
            traceItems.push({ kind: "thought", text: block.text.trim() });
            finalText += block.text + "\n";
          }
        } else if (block.type === "server_tool_use" && block.name === "web_search") {
          traceItems.push({
            kind: "search",
            query: (block.input && block.input.query) || "..."
          });
        } else if (block.type === "web_search_tool_result") {
          const results = Array.isArray(block.content) ? block.content : [];
          const sources = results
            .slice(0, 4)
            .map(r => r.title || r.url || "")
            .filter(Boolean);
          traceItems.push({
            kind: "results",
            count: results.length,
            sources
          });
        }
      }

      setTrace(traceItems);
      setRawOutput(finalText);

      const jsonMatch = finalText.match(/```json\s*([\s\S]+?)\s*```/);
      if (jsonMatch) {
        try {
          const parsed = JSON.parse(jsonMatch[1]);
          setResult(parsed);
        } catch (e) {
          setError("JSON parse failed. See raw output below.");
        }
      } else {
        setError("Agent did not produce a JSON block. See raw output below.");
      }
    } catch (e) {
      setError(e.message || String(e));
    } finally {
      setLoading(false);
    }
  }

  function formatUSD(n) {
    if (n == null || isNaN(n)) return "—";
    if (n === 0) return "$0";
    if (n >= 1e9) return `$${(n / 1e9).toFixed(2)}B`;
    if (n >= 1e6) return `$${(n / 1e6).toFixed(1)}M`;
    if (n >= 1e3) return `$${(n / 1e3).toFixed(0)}K`;
    return `$${n}`;
  }

  function confidenceColor(c) {
    if (c === "high") return "#40BECC";
    if (c === "medium") return "#c4a85a";
    return "#a07a7a";
  }

  function weightColor(w) {
    if (w === "high") return "#40BECC";
    if (w === "medium") return "#a8a59c";
    return "#5a5853";
  }

  const inputStyle = {
    width: '100%',
    background: 'transparent',
    border: 'none',
    borderBottom: '1px solid #2a2a2e',
    color: '#f4f2eb',
    padding: '12px 0',
    fontSize: '15px',
    fontFamily: '"Inter", -apple-system, sans-serif',
    outline: 'none',
    boxSizing: 'border-box'
  };

  return (
    <div style={{
      minHeight: '100vh',
      background: '#0a0a0c',
      backgroundImage: 'radial-gradient(ellipse 60% 40% at top, #14141822 0%, transparent 60%)',
      color: '#e8e6df',
      fontFamily: '"Inter", -apple-system, sans-serif',
      padding: '56px 24px 96px',
      fontSize: '14px',
      lineHeight: 1.6,
    }}>
      <div style={{ maxWidth: '720px', margin: '0 auto' }}>

        {/* HEADER */}
        <header style={{ marginBottom: '56px', borderBottom: '1px solid #1c1c20', paddingBottom: '36px' }}>
          <div style={{
            fontFamily: '"JetBrains Mono", monospace',
            fontSize: '10px',
            letterSpacing: '0.25em',
            color: '#40BECC',
            marginBottom: '14px',
            textTransform: 'uppercase'
          }}>
            module · signal.revenue_inference · v0
          </div>
          <h1 style={{
            fontFamily: '"Cormorant Garamond", serif',
            fontSize: '48px',
            fontWeight: 500,
            margin: 0,
            letterSpacing: '-0.02em',
            color: '#f4f2eb',
            lineHeight: 1.05
          }}>
            Revenue Inference Agent
          </h1>
          <p style={{
            fontFamily: '"Cormorant Garamond", serif',
            fontStyle: 'italic',
            fontSize: '19px',
            color: '#8a8780',
            margin: '12px 0 0',
            fontWeight: 400
          }}>
            Behavioral signals from the web, triangulated into a revenue range.
          </p>
        </header>

        {/* INPUT */}
        <section style={{ marginBottom: '56px' }}>
          <Label>Target brand</Label>
          <input
            type="text"
            value={brand}
            onChange={(e) => setBrand(e.target.value)}
            placeholder="e.g. MUD\WTR, Halo Collar, OOFOS"
            disabled={loading}
            onKeyDown={(e) => { if (e.key === 'Enter' && !loading) investigate(); }}
            style={inputStyle}
          />

          <div style={{ marginTop: '28px' }}>
            <Label>Context (optional)</Label>
            <input
              type="text"
              value={hint}
              onChange={(e) => setHint(e.target.value)}
              placeholder="domain, industry, geography, founding year…"
              disabled={loading}
              onKeyDown={(e) => { if (e.key === 'Enter' && !loading) investigate(); }}
              style={inputStyle}
            />
          </div>

          <button
            onClick={investigate}
            disabled={loading || !brand.trim()}
            style={{
              marginTop: '32px',
              background: loading || !brand.trim() ? 'transparent' : '#40BECC',
              color: loading || !brand.trim() ? '#5a5853' : '#0a0a0c',
              border: loading || !brand.trim() ? '1px solid #2a2a2e' : '1px solid #40BECC',
              padding: '14px 36px',
              fontFamily: '"JetBrains Mono", monospace',
              fontSize: '11px',
              letterSpacing: '0.18em',
              textTransform: 'uppercase',
              fontWeight: 500,
              cursor: loading || !brand.trim() ? 'not-allowed' : 'pointer',
              transition: 'all 0.15s ease',
            }}
          >
            {loading ? '· · · investigating' : '→ investigate'}
          </button>
        </section>

        {/* TRACE */}
        {(loading || trace.length > 0) && (
          <section style={{ marginBottom: '56px' }}>
            <SectionHead>Investigation log</SectionHead>
            <div style={{
              fontFamily: '"JetBrains Mono", monospace',
              fontSize: '12px',
              color: '#a8a59c',
            }}>
              {trace.map((item, i) => (
                <TraceItem key={i} index={i + 1} item={item} />
              ))}
              {loading && (
                <div style={{
                  color: '#40BECC',
                  padding: '14px 0',
                  fontStyle: 'italic',
                  fontSize: '12px',
                  letterSpacing: '0.05em'
                }}>
                  · · · agent working
                </div>
              )}
            </div>
          </section>
        )}

        {/* ERROR */}
        {error && (
          <div style={{
            background: '#1a1212',
            border: '1px solid #3a2222',
            padding: '16px 20px',
            fontFamily: '"JetBrains Mono", monospace',
            fontSize: '12px',
            color: '#d4a8a8',
            marginBottom: '36px'
          }}>
            ⚠ {error}
          </div>
        )}

        {/* RESULT */}
        {result && (
          <section>
            <SectionHead>Estimate</SectionHead>

            <div style={{
              border: '1px solid #2a2a2e',
              padding: '36px',
              marginBottom: '40px',
              background: 'linear-gradient(180deg, #0e0e12 0%, #0a0a0c 100%)'
            }}>
              <div style={{
                fontFamily: '"JetBrains Mono", monospace',
                fontSize: '10px',
                color: '#7a7770',
                letterSpacing: '0.22em',
                textTransform: 'uppercase',
                marginBottom: '16px'
              }}>
                annual revenue range
              </div>
              <div style={{
                fontFamily: '"Cormorant Garamond", serif',
                fontSize: '46px',
                color: '#f4f2eb',
                fontWeight: 500,
                letterSpacing: '-0.02em',
                lineHeight: 1.1
              }}>
                {formatUSD(result.revenue_estimate?.low)}
                <span style={{ color: '#40BECC', margin: '0 18px', fontSize: '32px', verticalAlign: 'middle' }}>—</span>
                {formatUSD(result.revenue_estimate?.high)}
              </div>
              <div style={{
                fontFamily: '"JetBrains Mono", monospace',
                fontSize: '12px',
                color: '#a8a59c',
                marginTop: '20px',
                display: 'flex',
                gap: '32px',
                flexWrap: 'wrap'
              }}>
                <span>central · <span style={{ color: '#f4f2eb' }}>{formatUSD(result.revenue_estimate?.central)}</span></span>
                <span>confidence · <span style={{ color: confidenceColor(result.confidence) }}>{result.confidence}</span></span>
              </div>
            </div>

            {result.signals_found && result.signals_found.length > 0 && (
              <>
                <SectionHead>Signals captured</SectionHead>
                <div style={{ marginBottom: '40px' }}>
                  {result.signals_found.map((s, i) => (
                    <SignalRow key={i} signal={s} weightColor={weightColor} />
                  ))}
                </div>
              </>
            )}

            {result.reasoning_summary && (
              <>
                <SectionHead>Reasoning</SectionHead>
                <p style={{
                  fontFamily: '"Cormorant Garamond", serif',
                  fontStyle: 'italic',
                  fontSize: '19px',
                  lineHeight: 1.55,
                  color: '#c8c5bc',
                  margin: '0 0 36px',
                  fontWeight: 400
                }}>
                  "{result.reasoning_summary}"
                </p>
              </>
            )}
          </section>
        )}

        {/* RAW OUTPUT */}
        {rawOutput && (
          <section style={{ marginTop: '48px', borderTop: '1px solid #1c1c20', paddingTop: '24px' }}>
            <button
              onClick={() => setShowRaw(!showRaw)}
              style={{
                background: 'transparent',
                border: 'none',
                color: '#5a5853',
                fontFamily: '"JetBrains Mono", monospace',
                fontSize: '10px',
                letterSpacing: '0.2em',
                textTransform: 'uppercase',
                cursor: 'pointer',
                padding: 0
              }}
            >
              {showRaw ? '▾ hide raw output' : '▸ show raw output'}
            </button>
            {showRaw && (
              <pre style={{
                marginTop: '16px',
                padding: '16px',
                background: '#0e0e12',
                border: '1px solid #1c1c20',
                fontFamily: '"JetBrains Mono", monospace',
                fontSize: '11px',
                color: '#8a8780',
                whiteSpace: 'pre-wrap',
                lineHeight: 1.5,
                maxHeight: '420px',
                overflowY: 'auto'
              }}>
                {rawOutput}
              </pre>
            )}
          </section>
        )}
      </div>
    </div>
  );
}

function Label({ children }) {
  return (
    <div style={{
      fontFamily: '"JetBrains Mono", monospace',
      fontSize: '10px',
      letterSpacing: '0.2em',
      color: '#7a7770',
      textTransform: 'uppercase',
      marginBottom: '6px'
    }}>
      {children}
    </div>
  );
}

function SectionHead({ children }) {
  return (
    <div style={{
      fontFamily: '"JetBrains Mono", monospace',
      fontSize: '10px',
      letterSpacing: '0.25em',
      color: '#40BECC',
      textTransform: 'uppercase',
      marginBottom: '20px',
      display: 'flex',
      alignItems: 'center',
      gap: '12px'
    }}>
      <span>{children}</span>
      <span style={{ flex: 1, height: '1px', background: '#1c1c20' }}></span>
    </div>
  );
}

function TraceItem({ index, item }) {
  const indexStyle = {
    color: '#5a5853',
    width: '40px',
    flexShrink: 0,
    fontSize: '11px'
  };

  const rowStyle = {
    display: 'flex',
    padding: '6px 0',
    gap: '4px',
    alignItems: 'baseline'
  };

  if (item.kind === "search") {
    return (
      <div style={rowStyle}>
        <span style={indexStyle}>[{String(index).padStart(2, '0')}]</span>
        <span style={{ color: '#40BECC' }}>search</span>
        <span style={{ color: '#5a5853', margin: '0 6px' }}>›</span>
        <span style={{ color: '#e8e6df' }}>"{item.query}"</span>
      </div>
    );
  }

  if (item.kind === "results") {
    return (
      <div style={rowStyle}>
        <span style={indexStyle}>[{String(index).padStart(2, '0')}]</span>
        <span style={{ color: '#8a8780' }}>fetched</span>
        <span style={{ color: '#5a5853', margin: '0 6px' }}>›</span>
        <span style={{ color: '#a8a59c' }}>
          {item.count} result{item.count === 1 ? '' : 's'}
          {item.sources && item.sources.length > 0 && (
            <span style={{ color: '#5a5853', marginLeft: '8px', fontStyle: 'italic' }}>
              ({item.sources.slice(0, 2).join(' · ').slice(0, 90)})
            </span>
          )}
        </span>
      </div>
    );
  }

  // thought
  const text = item.text.length > 320 ? item.text.slice(0, 320) + '…' : item.text;
  return (
    <div style={{ ...rowStyle, padding: '12px 0' }}>
      <span style={indexStyle}>[{String(index).padStart(2, '0')}]</span>
      <span style={{
        color: '#c8c5bc',
        fontFamily: '"Inter", -apple-system, sans-serif',
        fontSize: '13px',
        fontStyle: 'italic',
        lineHeight: 1.6
      }}>
        {text}
      </span>
    </div>
  );
}

function SignalRow({ signal, weightColor }) {
  return (
    <div style={{
      display: 'flex',
      padding: '14px 0',
      borderBottom: '1px solid #18181c',
      alignItems: 'flex-start',
      gap: '16px',
      fontFamily: '"JetBrains Mono", monospace',
      fontSize: '12px'
    }}>
      <div style={{
        width: '110px',
        flexShrink: 0,
        color: '#7a7770',
        textTransform: 'lowercase',
        paddingTop: '2px'
      }}>
        {signal.type}
      </div>
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ color: '#f4f2eb', fontWeight: 500 }}>
          {signal.label}
        </div>
        <div style={{ color: '#a8a59c', marginTop: '4px', fontSize: '11px', wordBreak: 'break-word' }}>
          {signal.value}
        </div>
        {signal.source && (
          <div style={{ color: '#5a5853', marginTop: '4px', fontSize: '10px', fontStyle: 'italic' }}>
            via {signal.source}
          </div>
        )}
      </div>
      <div style={{
        flexShrink: 0,
        color: weightColor(signal.weight),
        fontSize: '10px',
        letterSpacing: '0.15em',
        textTransform: 'uppercase',
        padding: '3px 8px',
        border: `1px solid ${weightColor(signal.weight)}44`
      }}>
        {signal.weight}
      </div>
    </div>
  );
}

export default App;
